# The upstream inline corpus parses each example as bare inline text.  These
# examples start with a line that is a block construct in a whole document,
# which is what tree-sitter-okf parses (CommonMark: an HTML block).

inline/spec.txt: Example 327 - https://github.github.com/gfm/#example-327  # a complete open tag alone on a line is an HTML block (type 7) in a document

# The upstream corpus is generated with every extension on.  OKF v0.2 has no
# `#tag` syntax, so the tags dialect is off (spec D7).
inline/spec.txt: Example 324 - https://github.github.com/gfm/#example-324  # `#;` and `#x;` are tags only in the tags dialect (spec D7)
inline/spec.txt: Example 638 - https://github.github.com/gfm/#example-638  # `#ref` is a tag only in the tags dialect (spec D7)
